**Goal:** The endotoxin section's footer text on the COA (PDF, verify page, badge) shows the result's LOD through a `{lod}` macro the lab writes into the analysis profile's text. An optional `[ ... ]` segment drops out when there is no LOD.

**Architecture:** A pure resolver module (`backend/coa/text_macros.py`) fills macros and optional segments. Mk1's native-sections wire build calls it per section with the section rows' LODs, so every downstream surface receives finished text and nothing outside Mk1 changes. The analysis-profile routes refuse unknown macros on save, and the profile editor shows a one-line hint.

**Tech Stack:** FastAPI + SQLAlchemy 2 (SQLite in unit tests, host dev Postgres in route tests), pytest; React 19 + TypeScript, vitest + Testing Library; npm only.

**Spec:** `docs/superpowers/specs/2026-10-13-endo-slice4-coa-text-macros-design.md` (approved 2026-10-13). Read it before any task.

```python
"""COA text macros (endo slice 4, spec
docs/superpowers/specs/2026-10-13-endo-slice4-coa-text-macros-design.md).

An analysis profile's COA text (basis note, method, prep, footnote text) may
carry macros such as {lod}. Mk1 fills them from the section's results when it
builds the COA wire, so the PDF, the verify page and the badge print finished
text. A [ ... ] segment that holds a known macro disappears when any known
macro in it has no value. Text with no known macro is returned exactly as
typed, so every existing profile text is unchanged. Pure: no database."""
from __future__ import annotations

import math
import re
from typing import Iterable, Mapping, Optional

MACROS: tuple[str, ...] = ("lod",)

_MACRO = re.compile(r"\{([a-z_]+)\}")
# A segment is a bracket pair with no bracket inside it, so segments never nest.
_SEGMENT = re.compile(r"\[([^\[\]]*)\]")
_SPACES = re.compile(r" {2,}")


def unknown_macros(text: Optional[str]) -> list[str]:
    """Macro names in text that Mk1 does not know, in order, each once."""
    out: list[str] = []
    for name in _MACRO.findall(text or ""):
        if name not in MACROS and name not in out:
            out.append(name)
    return out


def format_number(v: float) -> str:
    """Up to 6 decimals, no trailing zeros, no exponent: 0.005, 0.05, 25."""
    s = f"{v:.6f}".rstrip("0").rstrip(".")
    return s if s not in ("", "-0") else "0"


def format_lod(limits: Iterable[tuple[Optional[float], Optional[str]]]) -> str:
    """The section's {lod}: each row's "<number> <unit>", distinct, in row
    order, joined with ", ". Rows without a finite LOD are skipped."""
    out: list[str] = []
    for value, unit in limits:
        if value is None or not math.isfinite(value):
            continue
        text = format_number(value)
        if unit and unit.strip():
            text = f"{text} {unit.strip()}"
        if text not in out:
            out.append(text)
    return ", ".join(out)


def _has_known_macro(text: str) -> bool:
    return any(name in MACROS for name in _MACRO.findall(text))


def resolve(text: Optional[str], values: Mapping[str, str]) -> Optional[str]:
    """Fill known macros from values (a missing or empty value is empty),
    drop optional segments whose known macros are not all filled, and tidy
    spaces. Text with no known macro is returned exactly as typed; text that
    resolves to nothing becomes None."""
    if text is None or not _has_known_macro(text):
        return text

    def fill(m: re.Match) -> str:
        name = m.group(1)
        return (values.get(name) or "") if name in MACROS else m.group(0)

    def segment(m: re.Match) -> str:
        inner = m.group(1)
        names = [n for n in _MACRO.findall(inner) if n in MACROS]
        if not names:
            return m.group(0)
        if any(not values.get(n) for n in names):
            return ""
        return _MACRO.sub(fill, inner)

    out = _SEGMENT.sub(segment, text)
    out = _MACRO.sub(fill, out)
    out = "\n".join(_SPACES.sub(" ", line).strip() for line in out.split("\n")).strip()
    return out or None
```

```python
def _section_footnotes(raw, values: dict) -> list:
    """The profile's footnotes with macros resolved (endo slice 4). A note
```

```python
def _validate_coa_text_macros(fields: dict) -> None:
    """Endo slice 4: COA text may carry only the macros coa.text_macros knows,
    so a typo never reaches a certificate (the COA build itself leaves an
    unknown macro as typed)."""
    from coa.text_macros import MACROS, unknown_macros
    texts = [fields.get(k) for k in ("coa_basis_note", "coa_method_text", "coa_prep_text")]
    notes = fields.get("coa_footnotes")
    if isinstance(notes, list):
        texts += [n.get("text") for n in notes if isinstance(n, dict)]
    for text in texts:
        bad = unknown_macros(text if isinstance(text, str) else None)
        if bad:
            supported = ", ".join("{" + m + "}" for m in MACROS)
            raise HTTPException(
                400, "unknown COA text macro {" + bad[0] + "}; supported: " + supported)
```
